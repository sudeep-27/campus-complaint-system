2. CLIENT-SIDE INTERACTION SCRIPT (Javascript / AJAX Live Refresh)
This snippet from "dashboard.js" defines the "DashboardStats" class, which fetches the live stats endpoint asynchronously, animates the numerical indicators, and updates the conic-gradient style of the donut chart dynamically without reloading the web page.

class DashboardStats {
  // Fetch live counts from backend API and update DOM elements
  static refresh() {
    fetch('/api/dashboard-stats')
      .then(response => response.json())
      .then(data => {
        if (data.success) {
          const stats = data.stats;
          
          // 1. Animate stat numbers to new values
          this.animateValue('statTotal', stats.total);
          this.animateValue('statPending', stats.pending);
          this.animateValue('statInProgress', stats.in_progress);
          this.animateValue('statResolved', stats.resolved);
          
          // 2. Update CSS Conic Gradient Donut Chart
          const donutEl = document.getElementById('statDonut');
          const textEl = document.getElementById('statDonutText');
          if (donutEl && textEl) {
            const rate = data.resolution_rate || 0;
            const degrees = (rate / 100) * 360;
            
            // Re-render CSS background style dynamically
            donutEl.style.background = `conic-gradient(#10b981 ${degrees}deg, #1f2937 ${degrees}deg)`;
            textEl.textContent = `${rate}%`;
          }
        }
      })
      .catch(error => console.error('Error loading live dashboard stats:', error));
  }

  // Smooth integer counter animation
  static animateValue(id, targetValue) {
    const el = document.getElementById(id);
    if (!el) return;
    const startValue = parseInt(el.textContent) || 0;
    if (startValue === targetValue) return;
    
    const duration = 800; // Animation duration in milliseconds
    const startTime = performance.now();
    
    function update(currentTime) {
      const elapsed = currentTime - startTime;
      const progress = Math.min(elapsed / duration, 1);
      const currentValue = Math.floor(startValue + progress * (targetValue - startValue));
      el.textContent = currentValue;
      if (progress < 1) {
        requestAnimationFrame(update);
      }
    }
    requestAnimationFrame(update);
  }
}
